 package seng201.nunbutblood.services;
 
 import seng201.nunbutblood.models.Saint;
 import seng201.nunbutblood.models.StatType;
 
 import java.util.List;
 import java.util.Random;
 
 /**
  * The type Mid crusade events.
  */
 public class MidCrusadeEvents {
 
     private final Random random = new Random();
 
     /**
      * The type Choice.
      */
     public record Choice(
             String text,
             StatType stat,
             int amount,
             Saint saint) {}
 
     /**
      * The type Event.
      */
     public record Event(
             String description,
             Choice choiceA,
             Choice choiceB
     ) {}
 
     /**
      * Gets random event.
      *
      * @return the random event
      */
     public Event getRandomEvent(List<Saint> activeParty) {
 
         // Possiblity for same
         Saint target = activeParty.get(random.nextInt(activeParty.size()));
         Saint target2 = activeParty.get(random.nextInt(activeParty.size()));
 
         return switch (random.nextInt(1)) {
             case 0 -> new Event(
                     "The Demon's powerful stomp shakes the very earth you stand on! ",
 
                     new Choice(String.format("%s dives to save the team, leading them to take -%d hit" +
                             " to their %s", target, 2, StatType.STRENGTH), StatType.STRENGTH, -2, target),
 
                     new Choice(String.format("%s dives to save the team, leading them to take -%d hit" +
                     " to their %s", target2, -2, StatType.DEXTERITY), StatType.DEXTERITY, -2, target2)
             );
             default -> null;
 
         };
     }
 
     /**
      * Apply choice string.
      *
      * @param party  the party
      * @param choice the choice
      * @return the string
      */
     public String applyChoice(List<Saint> party, Choice choice) {
         if (party.isEmpty()) {
             return "No saints were affected.";
         }
 
         Saint target = party.get(random.nextInt(party.size()));
         applyStat(target, choice.stat(), choice.amount());
 
         String changeText = choice.amount() > 0 ? "+" + choice.amount() : String.valueOf(choice.amount());
 
         return target.getName() + "'s " + choice.stat().label + " changed by " + changeText + ".";
     }
 
     private void applyStat(Saint saint, StatType stat, int amount) {
         Saint.applyStat(saint, stat, amount);
     }
 }
